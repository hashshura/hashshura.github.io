_It's been over ten years since I first stepped into high school._

___

### Waktu...

Waktu berjalan cepat. Rasanya, baru kemarin saya mengenakan seragam putih abu-abu untuk pertama kalinya, mengikuti Masa Orientasi Siswa. Tiga hari yang melelahkan, penuh tugas yang menghabiskan waktu 4-6 jam setiap malam, setelah seharian penuh kegiatan yang baru selesai menjelang shalat Isya.

> _“Keputusan kepala sekolah: karena kalian belum dianggap layak jadi siswa SMA, Masa Orientasi Siswa diperpanjang satu hari lagi!”_

<div class="toy" data-toy="cry" data-h="190"></div>

Candaan itu masih melekat di ingatan saya, meski hanya sebuah lelucon untuk menambah sensasi.

___

Waktu berjalan sangat cepat. _I used to be pretty good at school_. Sebagai siswa SD, mendapatkan nilai tinggi bukanlah masalah besar. Dengarkan guru di kelas, jangan berbuat aneh-aneh, dan sisanya bersenang-senang di rumah dengan PlayStation 2. CD/DVD bajakan? Pasti. Soalnya, memang itu yang ada.

Saya juga ingat sore-sore santai di rumah. Membaca komik Doraemon sambil meneguk segelas es teh manis — sebenarnya dua gelas, karena keluarga kami memang sangat menyukai teh manis.

<div class="toy" data-toy="esteh" data-h="190"></div>

Waktu masih terasa cepat. Sabtu sore kemarin, kami sekeluarga mampir ke kafe bertema makanan Jepang. Saya hanya memesan _ocha_ tawar. Bukan karena sedang berdiet atau apa, tapi mungkin karena sekarang saya lebih sadar pentingnya mengurangi gula. Atau mungkin… es teh manis sudah tidak seenak dulu lagi.

### ...berjalan...

_During college, I used to write blogs—until everything was wiped out, burnt to ashes by my cloud provider due to an issue on their end. In return, I was compensated with a measly one year of free hosting. The rest of my posts? Irrecoverable. Led me to explore GitHub Pages and ultimately, this blog was born._

Masih ada mimpi-mimpi yang belum tercapai, sebagian karena saya tidak benar-benar mengejarnya.

(function(){

        ctx.stroke();
        ctx.lineWidth = 1.6;
        for (var i=0;i<5;i++){

        }
        // flames
        ctx.strokeStyle = '#e0632a'; ctx.lineWidth = 2.2;
        var f = Math.max(0, s.fire);
        for (var k=0;k<3;k++){
          var fx = cx - 22 + k*22, hgt = (26 + Math.sin(s.t*0.2 + k)*5) * f;
          ctx.beginPath();
          ctx.moveTo(fx-9, py-ph+2);
          ctx.quadraticCurveTo(fx-4, py-ph-hgt*0.6, fx, py-ph-hgt);
          ctx.quadraticCurveTo(fx+5, py-ph-hgt*0.5, fx+9, py-ph+2);
          ctx.stroke();
        }
      } else {
        // rebuilt: a small browser window
        ink(ctx);
        ctx.strokeRect(cx-72, py-72, 144, 72);
        ctx.beginPath(); ctx.moveTo(cx-72, py-56); ctx.lineTo(cx+72, py-56); ctx.stroke();
        ctx.beginPath();
        ctx.arc(cx-62, py-64, 2.6, 0, 7); ctx.arc(cx-52, py-64, 2.6, 0, 7); ctx.arc(cx-42, py-64, 2.6, 0, 7);
        ctx.stroke();
        ctx.lineWidth = 1.6;
        ctx.beginPath();
        ctx.moveTo(cx-60, py-42); ctx.lineTo(cx+40, py-42);
        ctx.moveTo(cx-60, py-30); ctx.lineTo(cx+16, py-30);
        ctx.moveTo(cx-60, py-18); ctx.lineTo(cx+52, py-18);
        ctx.stroke();
        txt(ctx, 'hashshura.github.io', cx + 22, py-59, 10, INK, true);
      }
      // water drops
      ctx.strokeStyle = '#3b7ea1'; ctx.lineWidth = 2;
      for (var d=0;d<s.drops.length;d++){
        var dr = s.drops[d];
        ctx.beginPath(); ctx.moveTo(dr.x, dr.y); ctx.lineTo(dr.x, dr.y+5); ctx.stroke();
      }
      // rising ash
      ctx.fillStyle = '#999';
      for (var a=0;a<s.ash.length;a++){
        var as = s.ash[a];
        ctx.globalAlpha = Math.max(0, as.life/90);
        ctx.fillRect(as.x, as.y, 2, 2);
      }
      ctx.globalAlpha = 1;
      txt(ctx, s.done ? 'Blog ini' : 'Blog lama, 2018', W/2, 22, 13, INK, true);
      var note = s.done ? 'abunya jadi halaman yang sedang kamu baca.'
               : s.fire>0.5 ? 'gantinya: hosting gratis satu tahun.'
               : 'hampir padam. tulisannya tetap hilang.';
      txt(ctx, note, W/2, H-12, 12);
    }
  };

  // 7. Ketindihan: the shadow leans closer — until it turns out to be a cat
  TOYS.tindih = {
    init: function(){ return { lvl:0, t:0 }; },
    click: function(s){ if (s.lvl<5) s.lvl++; },
    tick: function(s){ s.t++; },
    draw: function(ctx, W, H, s){
      var bx = W*0.5, by = H-42;
      ink(ctx);
      // bed
      ctx.beginPath();
      ctx.moveTo(bx-86, by); ctx.lineTo(bx+86, by);
      ctx.moveTo(bx-86, by); ctx.lineTo(bx-86, by+16);
      ctx.moveTo(bx+86, by); ctx.lineTo(bx+86, by+16);
      ctx.stroke();
      // blanket + head on pillow
      ctx.beginPath();
      ctx.moveTo(bx-58, by); ctx.quadraticCurveTo(bx-20, by-26, bx+34, by-14); ctx.lineTo(bx+70, by-12);
      ctx.stroke();
      ctx.beginPath(); ctx.arc(bx-70, by-14, 13, 0, 7); ctx.stroke();
      // wide-awake eyes
      ctx.beginPath();
      ctx.arc(bx-74, by-17, 2.2, 0, 7); ctx.arc(bx-65, by-17, 2.2, 0, 7); ctx.fill();

        ctx.closePath(); ctx.fill();

        ctx.fillStyle = PAPER;
        ctx.beginPath();
        ctx.arc(sx-5, sy-sh+6, 2.4, 0, 7); ctx.arc(sx+5, sy-sh+6, 2.4, 0, 7); ctx.fill();
        ctx.restore();

        // it was a cat sitting on your chest the whole time

        ctx.beginPath(); ctx.ellipse(kx, ky-6, 19, 12, 0, 0, 7); ctx.stroke();
        ctx.beginPath(); ctx.arc(kx+16, ky-18, 11, 0, 7); ctx.stroke();

  };

  // 8. Waktu: try to stop the clock; you can't, but it does calm down
  TOYS.jam = {
    init: function(){ return { ang:0, spd:0.16, clicks:0, calm:false }; },
    click: function(s){
      s.clicks++;
      s.spd *= 0.22;
      if (s.clicks >= 3) s.calm = true;
    },
    tick: function(s){
      if (!s.calm) s.spd += (0.16 - s.spd)*0.02; else s.spd += (0.02 - s.spd)*0.04;
      s.ang += s.spd;
    },
    draw: function(ctx, W, H, s){
      var cx = W/2, cy = H/2 + 6, R = Math.min(52, H*0.28);
      ink(ctx);
      ctx.beginPath(); ctx.arc(cx, cy, R, 0, 7); ctx.stroke();
      ctx.lineWidth = 1.7;
      for (var i=0;i<12;i++){
        var a = i*Math.PI/6;
        ctx.beginPath();
        ctx.moveTo(cx+Math.cos(a)*(R-4), cy+Math.sin(a)*(R-4));
        ctx.lineTo(cx+Math.cos(a)*(R-9), cy+Math.sin(a)*(R-9));
        ctx.stroke();
      }
      ink(ctx, 2.6);
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.lineTo(cx+Math.cos(s.ang-Math.PI/2)*(R-14), cy+Math.sin(s.ang-Math.PI/2)*(R-14));
      ctx.stroke();
      ink(ctx, 3);
      ctx.beginPath();
      ctx.moveTo(cx, cy);
      ctx.lineTo(cx+Math.cos(s.ang/12-Math.PI/2)*(R-26), cy+Math.sin(s.ang/12-Math.PI/2)*(R-26));
      ctx.stroke();
      ctx.beginPath(); ctx.arc(cx, cy, 3, 0, 7); ctx.fill();
      var year = 2025 + Math.floor(s.ang/(Math.PI*2*12));
      txt(ctx, String(year), W/2, 24, 15, INK, true);
      var note = s.calm ? 'tidak bisa distop. tapi kamu boleh duduk dulu.'
               : s.clicks===0 ? 'klik kalau mau coba menahannya.'
               : s.clicks===1 ? 'melambat sebentar…'
               : 'lalu jalan lagi, seperti biasa.';
      txt(ctx, note, W/2, H-14, 12);
    }
  };

  // 9. Penutup: a stickman who waves back
  TOYS.dadah = {
    init: function(){ return { e:0.15, t:0, hearts:[] }; },
    click: function(s, W, H){
      s.e = Math.min(1, s.e + 0.42);
      for (var i=0;i<5;i++) s.hearts.push({ x: W/2 + rnd(-26,26), y: H-90, vy: rnd(-1.6,-0.7), life: 60 });
    },
    tick: function(s){
      s.t++; s.e = Math.max(0.15, s.e - 0.0035);
      for (var i=s.hearts.length-1;i>=0;i--){ var h=s.hearts[i]; h.y+=h.vy; h.life--; if (h.life<=0) s.hearts.splice(i,1); }
    },
    draw: function(ctx, W, H, s){
      var cx = W/2, ground = H-30, head = H-104;
      ink(ctx);
      var bob = Math.sin(s.t*0.1*(0.5+s.e))*s.e*2;
      ctx.save(); ctx.translate(0, bob);
      ctx.beginPath();
      ctx.moveTo(cx, head+15); ctx.lineTo(cx, ground-26);
      ctx.moveTo(cx, ground-26); ctx.lineTo(cx-12, ground);
      ctx.moveTo(cx, ground-26); ctx.lineTo(cx+12, ground);
      ctx.moveTo(cx, head+26); ctx.lineTo(cx-17, head+34);
      ctx.stroke();
      // waving arm: shoulder → elbow → hand, up above the head
      var wav = Math.sin(s.t*0.22*(0.6+s.e*1.6)) * (0.3 + s.e*0.7);
      var ex = cx + 16, ey = head + 12;
      ctx.beginPath();
      ctx.moveTo(cx, head+26);
      ctx.lineTo(ex, ey);
      ctx.lineTo(ex + 8 + wav*10, ey - 24 - Math.abs(wav)*4);
      ctx.stroke();
      // open palm
      ctx.beginPath();
      ctx.arc(ex + 8 + wav*10, ey - 26 - Math.abs(wav)*4, 3.4, 0, 7);
      ctx.stroke();
      // head + smile
      ctx.beginPath(); ctx.arc(cx, head, 15, 0, 7); ctx.stroke();
      ctx.beginPath();
      ctx.arc(cx-5, head-4, 1.7, 0, 7); ctx.arc(cx+5, head-4, 1.7, 0, 7); ctx.fill();
      ctx.beginPath(); ctx.arc(cx, head+1, 8, 0.15*Math.PI, 0.85*Math.PI); ctx.stroke();
      ctx.restore();
      // hearts
      ctx.save(); ctx.textAlign='center'; ctx.fillStyle='#c0392b';
      for (var i=0;i<s.hearts.length;i++){

      txt(ctx, s.e>0.75 ? 'makasih sudah baca sampai sini.' : 'dia bakal balas kalau kamu duluan.', W/2, H-14, 12);
    }
  };

  // ------------------------------------------------------------ bootstrap ----
  var mounted = [];
  var nodes = document.querySelectorAll('.toy[data-toy]');
  for (var i=0;i<nodes.length;i++){
    (function(el){
      var def = TOYS[el.getAttribute('data-toy')];
      if (!def) return;
      var h = parseInt(el.getAttribute('data-h') || '180', 10);
      var cv = document.createElement('canvas');
      var hint = document.createElement('div');
      hint.className = 'hint'; hint.textContent = '👆 klik';
      el.appendChild(cv); el.appendChild(hint);
      var toy = { el:el, cv:cv, ctx:cv.getContext('2d'), def:def, h:h, s:def.init() };
      function size(){
        var w = Math.max(240, Math.min(460, el.clientWidth || 460));
        cv.width = Math.round(w); cv.height = h;
      }
      toy.size = size; size();
      function hit(e){
        if (e && e.preventDefault) e.preventDefault();
        el.classList.add('touched');
        def.click && def.click(toy.s, cv.width, cv.height);
      }
      cv.addEventListener('mousedown', hit);
      cv.addEventListener('touchstart', hit, {passive:false});
      mounted.push(toy);
    })(nodes[i]);
  }

  window.addEventListener('resize', function(){
    for (var i=0;i<mounted.length;i++) mounted[i].size();
  });

  function loop(){
    var vh = window.innerHeight || 800;
    for (var i=0;i<mounted.length;i++){
      var toy = mounted[i];
      var r = toy.cv.getBoundingClientRect();
      if (r.bottom < -80 || r.top > vh + 80) continue; // offscreen: don't burn cycles
      var W = toy.cv.width, H = toy.cv.height;
      toy.def.tick && toy.def.tick(toy.s, W, H);
      toy.ctx.fillStyle = PAPER; toy.ctx.fillRect(0,0,W,H);
      toy.def.draw(toy.ctx, W, H, toy.s);
    }
    requestAnimationFrame(loop);
  }
  loop();

  // ---- EYD ↔ Jaksel toggle (text, not canvas) ----
  var swap = document.getElementById('toy-swap');
  if (swap){
    var modes = [
      { tag:'EYD', text:'Sesungguhnya saya harus pulang lebih awal hari ini karena ada urusan keluarga yang tidak dapat ditunda.' },
      { tag:'Jaksel', text:'Which is, gue harus cabut duluan, ada family thing yang literally nggak bisa di-postpone, sih.' }
    ];
    var m = 0;
    var tag = document.createElement('div'); tag.className = 'tag';
    var sen = document.createElement('div'); sen.className = 'sentence';
    var hint = document.createElement('div'); hint.className = 'hint'; hint.textContent = '👆 klik untuk ganti mode';
    swap.appendChild(tag); swap.appendChild(sen); swap.appendChild(hint);
    function render(){ tag.textContent = modes[m].tag; sen.textContent = modes[m].text; }
    render();
    swap.addEventListener('click', function(){ m = (m+1) % modes.length; render(); });
  }
})();
