        retries++;

        reconnectNote = 'Reconnecting…';
        // Hold the seat we are actually retrying. Leaving the room, or starting
        // a bot match, clears `joined` — so a timer that read it at fire time
        // would either throw or drag us back into a room we had left.
        var target = joined;
        retryTimer = setTimeout(function () {
          retryTimer = null;
          if (leaving || joined !== target) return;
          connect(target.code, target.pw, target.token);
        }, wait);
      } else {
        reconnectNote = '';
        showScreen('menu');
        noteEl.textContent = 'Disconnected.';
      }
    };
    ws.onerror = function () {};
  }

  function send(obj) {
    if (botMode) { handleLocalAction(obj); return; }
    if (ws && ws.readyState === 1) ws.send(JSON.stringify(obj));
  }

  // ---- play vs bot --------------------------------------------------------
  // No server involved at all: the exact same sim runs locally, bots "type"
  // on a timer instead of a real person tapping keys, and everything else —
  // rendering, the legend, auto-submit, animations — is the code already
  // built for the online game. A bot is just a different transport feeding
  // onState() the same shape of message a room's WebSocket would.
  var botMode = false, botWorld = null, botTimers = {};
  var BOT_NAMES = ['Nyx', 'Rook', 'Vex', 'Ori', 'Fenn'];
  var BOT_CPS = 50 * 5 / 60;   // 50 wpm at ~5 chars/word average

  function localStateFor(world, viewerId) {
    var now = Date.now();
    var players = {};
    for (var id in world.players) {
      var p = world.players[id];
      var mine = id === viewerId;
      var hideMe = !mine && S.isVanished(p, now);
      players[id] = {
        cls: p.cls, name: p.name, hp: p.hp, dead: p.dead,
        x: hideMe ? null : p.x, y: hideMe ? null : p.y,
        hidden: hideMe, stunned: S.isStunned(p, now),
        stunUntil: p.stunnedUntil || 0, stunSpan: p.stunSpan || 0
      };
    }
    var me = world.players[viewerId];
    return {
      type: 'state', you: viewerId, players: players,
      words: me && !me.dead ? me.words : null,
      specialReadyAt: me ? me.specialCooldownUntil : null,
      vanishUntil: me ? me.vanishUntil : 0,
      trap: me && world.traps[viewerId] ? world.traps[viewerId] : null
    };
  }

  function localBroadcast(event) {
    var payload = localStateFor(botWorld, 'me');
    if (event) payload.event = event;
    onState(payload);
    scheduleLocalRefresh();
    var winner = S.checkWinner(botWorld);
    if (winner !== undefined) { stopBots(); onEnded({ winner: winner }); }
  }

  // Same reason as the server's version: vanish and stun end on a clock, but
  // state only otherwise refreshes when somebody acts.
  var localRefreshTimer = null;
  function scheduleLocalRefresh() {
    if (localRefreshTimer) { clearTimeout(localRefreshTimer); localRefreshTimer = null; }
    if (!botMode || !botWorld) return;
    var now = Date.now(), next = Infinity;
    for (var id in botWorld.players) {
      var p = botWorld.players[id];
      if (p.vanishUntil > now) next = Math.min(next, p.vanishUntil);
      if (p.stunnedUntil > now) next = Math.min(next, p.stunnedUntil);
    }
    if (next === Infinity) return;
    localRefreshTimer = setTimeout(function () {
      localRefreshTimer = null;
      if (botMode) localBroadcast(null);
    }, Math.max(20, next - now + 25));
  }

  function handleLocalAction(obj) {
    if (obj.type !== 'action' || !botWorld) return;
    var result = S.resolveAction(botWorld, 'me', obj.slot, Date.now());
    if (result) localBroadcast(result);
  }

  function stopBots() {
    for (var id in botTimers) clearTimeout(botTimers[id]);
    botTimers = {};
    if (localRefreshTimer) { clearTimeout(localRefreshTimer); localRefreshTimer = null; }
  }

  var STEPS = [['moveUp', 0, -1], ['moveDown', 0, 1], ['moveLeft', -1, 0], ['moveRight', 1, 0]];

  // Breadth-first search for the first step of a real shortest path, because
  // stepping greedily toward the target does not work on a map with walls:
  // put a wall between bot and target and greedy stepping oscillates between
  // two tiles forever, which is exactly the "bots stand there doing nothing"
  // behaviour. Rivers block walking too, but map generation guarantees the
  // floor is connected without crossing one, so a path always exists.
  function botStepToward(world, p, tx, ty) {
    var startKey = p.x + ',' + p.y;
    var seen = {}, prev = {}, q = [[p.x, p.y]];
    seen[startKey] = true;
    while (q.length) {
      var cur = q.shift(), cx = cur[0], cy = cur[1];
      if (cx === tx && cy === ty) {
        var k = cx + ',' + cy;
        while (prev[k] && prev[k].from !== startKey) k = prev[k].from;
        return prev[k] ? prev[k].slot : null;
      }
      for (var i = 0; i < STEPS.length; i++) {
        var nx = cx + STEPS[i][1], ny = cy + STEPS[i][2];
        if (nx < 0 || ny < 0 || nx >= world.w || ny >= world.h) continue;
        var nk = nx + ',' + ny;
        if (seen[nk]) continue;
        var tile = world.grid[ny][nx];
        if (tile === S.WALL || tile === S.RIVER) continue;
        // The goal tile is occupied by the target, so allow it as a
        // destination but never as something to route through.
        if (world.occupancy[nk] && !(nx === tx && ny === ty)) continue;
        seen[nk] = true;
        prev[nk] = { from: cx + ',' + cy, slot: STEPS[i][0] };
        q.push([nx, ny]);
      }
    }
    return null;
  }

  // Which attack direction, if any, actually lands on this target — asked of
  // the sim's own shape logic rather than re-derived here, so a bot can never
  // think it has reach the real rules disagree with (fighter's swipe is wide,
  // ranger's shot is a line, the mage's beam is stopped by walls but not
  // rivers, and so on).
  function botAttackDir(world, p, targetId) {
    var shape = S.CLASSES[p.cls].attackShape;
    var dirs = ['up', 'down', 'left', 'right'];
    for (var i = 0; i < dirs.length; i++) {
      var swept = S.targetsInShape(world, p, dirs[i], shape);
      if (swept.hits.indexOf(targetId) !== -1) {
        return 'atk' + dirs[i].charAt(0).toUpperCase() + dirs[i].slice(1);
      }
    }
    return null;
  }

  // Hunt the nearest living target: shoot it if it is already in reach, use
  // the special when there is a concrete reason to, otherwise walk one step
  // along the shortest path to it. Deliberately simple — it is a practice
  // opponent, not a rival — but it does have to actually show up and fight.
  function decideBotAction(world, id) {
    var now = Date.now();
    var p = world.players[id];
    var def = S.CLASSES[p.cls];
    var targets = Object.keys(world.players).filter(function (oid) {
      return oid !== id && !world.players[oid].dead && !S.isVanished(world.players[oid], now);
    });
    if (!targets.length) return null;
    targets.sort(function (a, b) {
      var pa = world.players[a], pb = world.players[b];
      return (Math.abs(pa.x - p.x) + Math.abs(pa.y - p.y)) - (Math.abs(pb.x - p.x) + Math.abs(pb.y - p.y));
    });
    var tid = targets[0], t = world.players[tid];
    var gap = Math.abs(t.x - p.x) + Math.abs(t.y - p.y);

    if (p.specialCooldownUntil <= now) {
      if (def.special.kind === 'heal' && p.hp <= 60) return { slot: 'special' };
      if (def.special.kind === 'vanish' && p.hp <= 40 && gap <= 3) return { slot: 'special' };
      if (def.special.kind === 'shieldStun' &&
          S.targetsInShape(world, p, null, def.special.shape || 'self3x3').hits.length) return { slot: 'special' };
      // A trap only pays off on ground somebody might cross, so lay it while
      // the target is still a little way off rather than mid-brawl.
      if (def.special.kind === 'trap' && !world.traps[id] && gap > 2) return { slot: 'special' };
    }

    var atk = botAttackDir(world, p, tid);
    if (atk) return { slot: atk };

    var step = botStepToward(world, p, t.x, t.y);
    if (step) return { slot: step };

    // Boxed in for the moment (usually another body in a corridor) — shuffle
    // somewhere legal rather than freezing, so it never looks switched off.
    for (var i = 0; i < STEPS.length; i++) {
      var s = STEPS[(i + Math.floor(Math.random() * STEPS.length)) % STEPS.length];
      if (S.walkable(world, p.x + s[1], p.y + s[2])) return { slot: s[0] };
    }
    return null;
  }

  function scheduleBot(id) {
    if (!botMode) return;
    var p = botWorld.players[id];
    if (!p || p.dead) return;
    var decision = decideBotAction(botWorld, id);
    if (!decision) { botTimers[id] = setTimeout(function () { scheduleBot(id); }, 400); return; }
    var word = S.currentWord(p, decision.slot);
    var delayMs = (word.length / BOT_CPS) * 1000 + 150 + Math.random() * 200;
    botTimers[id] = setTimeout(function () {
      var result = S.resolveAction(botWorld, id, decision.slot, Date.now());
      if (result) localBroadcast(result);
      scheduleBot(id);
    }, delayMs);
  }

  // Turns one resolved-action event into: an updated facing direction (so the
  // character's accessory points the right way), a brief flash over every
  // tile the attack actually swept, a floating number per hit, and a small
  // pulse for whichever special just fired. Purely cosmetic — the event
  // already changed real state before this ever runs.
  function applyEventFx(ev) {
    var now = Date.now();
    var actor = players[ev.id];
    if (ev.slot && ev.slot.indexOf('move') === 0 && ev.ok) {
      facing[ev.id] = ev.slot.slice(4).toLowerCase();
    }
    if (ev.slot && ev.slot.indexOf('atk') === 0 && ev.dir) {
      facing[ev.id] = ev.dir;
      swings[ev.id] = { born: now, until: now + 230 };
    }
    if (ev.tiles && ev.tiles.length) {
      fx.push({ type: 'tiles', tiles: ev.tiles,
        color: ev.special === 'shieldStun' ? 'rgba(192,57,43,.45)' : 'rgba(241,196,15,.5)',
        until: now + 220 });
    }
    if (ev.hits) {
      ev.hits.forEach(function (h) {
        var t = players[h.id];
        if (t) fx.push({ type: 'dmg', x: t.x, y: t.y, text: '-' + h.dmg, born: now, until: now + 700 });
      });
    }
    if (ev.special === 'vanish' && actor) {
      fx.push({ type: 'pulse', x: actor.x, y: actor.y, color: 'rgba(142,68,173,.55)', born: now, until: now + 320 });
    }
    if (ev.special === 'heal' && actor) {
      fx.push({ type: 'pulse', x: actor.x, y: actor.y, color: 'rgba(46,139,87,.55)', born: now, until: now + 320 });
      fx.push({ type: 'dmg', x: actor.x, y: actor.y, text: '+heal', born: now, until: now + 700, good: true });
    }
    if (ev.trapPlaced && actor) {
      fx.push({ type: 'pulse', x: ev.trapPlaced.x, y: ev.trapPlaced.y, color: 'rgba(211,84,0,.5)', born: now, until: now + 320 });
    }
    if (ev.trapTriggered && actor) {
      fx.push({ type: 'pulse', x: actor.x, y: actor.y, color: 'rgba(211,84,0,.6)', born: now, until: now + 320 });
    }
  }

  // ---- typing ---------------------------------------------------------
  // A custom keyboard, not the OS one: on a phone the real keyboard eats
  // roughly half the screen and its height changes unpredictably as it
  // opens, so the map either disappears or jumps around mid-match. This one
  // has a fixed, known height, so the arena above it is always visible. The
  // input itself is `readonly` — tapping it never summons the real keyboard —
  // but a physical keyboard (desktop, or a phone with one paired) still works
  // via the document-level keydown listener below.
  //
  // The buffer decides its own target slot: whichever of my 9 live words it
  // exactly matches. Space/Enter/the on-screen space bar submits; a non-match
  // just clears, no penalty beyond the time spent typing it.
  var typingDisabled = false;

  function typeChar(c) {
    if (typingDisabled) return;
    typedBuf = (typedBuf + c).slice(0, 16);   // no live word is anywhere near this long
    typedEl.value = typedBuf;
    autoSubmitIfUnambiguous();
  }
  function backspace() {
    if (typingDisabled) return;
    typedBuf = typedBuf.slice(0, -1);
    typedEl.value = typedBuf;
    autoSubmitIfUnambiguous();
  }
  // "as" typed against {as, ash} still needs an explicit submit — it's an
  // exact match, but also a real prefix of a longer live word, so finishing
  // there could still mean "ash" is coming. Only fire early once the buffer
  // exactly matches a word AND no other live word extends past that same
  // buffer — e.g. typing the "h" in "ash" completes it unambiguously.
  function autoSubmitIfUnambiguous() {
    if (!myWords) return;
    var buf = typedBuf.trim().toLowerCase();
    if (!buf) return;
    var exact = null, ambiguous = false;
    for (var slot in myWords) {
      var w = (myWords[slot][0] || '').toLowerCase();
      if (!w) continue;
      if (w === buf) exact = slot;
      else if (w.length > buf.length && w.indexOf(buf) === 0) ambiguous = true;
    }
    if (exact && !ambiguous) submitTyped();
  }
  function submitTyped() {
    if (typingDisabled) return;
    var word = typedBuf.trim().toLowerCase();
    typedBuf = ''; typedEl.value = '';
    if (!word || !myWords) return;
    for (var slot in myWords) {
      if ((myWords[slot][0] || '').toLowerCase() === word) {
        send({ type: 'action', slot: slot, word: word });
        return;
      }
    }
    // no match: nothing happens, buffer is already cleared
  }

  function keySpacer(cls) {
    var s = document.createElement('span');
    s.className = 'wf-key pad ' + cls;
    return s;
  }

  // Touch and click are both handled on the container, and a press is resolved
  // to the NEAREST key rather than whatever element happens to sit under the
  // finger. Two reasons. First, the 4px gaps between keys and the invisible
  // spacers that produce the stagger were dead zones — a thumb landing between
  // s and d registered nothing. Second, two fingers landing at once arrive as a
  // single touchstart on whichever key was hit first, carrying both touch
  // points, so per-key listeners never hear about the second finger.
  //
  // Vertical distance is weighted heavier than horizontal, which makes the
  // outermost keys generous outwards: the space left of A, or the gap where a
  // phone would put shift next to Z, belongs to that row's edge key rather than
  // jumping to whatever sits diagonally above it.
  var keyEls = [];
  var ROW_BIAS = 1.35;

  function nearestKey(x, y) {
    var best = null, bestD = Infinity, bestH = 40;
    for (var i = 0; i < keyEls.length; i++) {
      var r = keyEls[i].getBoundingClientRect();
      if (!r.width) continue;
      var dx = x < r.left ? r.left - x : (x > r.right ? x - r.right : 0);
      var dy = (y < r.top ? r.top - y : (y > r.bottom ? y - r.bottom : 0)) * ROW_BIAS;
      var d = dx * dx + dy * dy;
      if (d < bestD) { bestD = d; best = keyEls[i]; bestH = r.height; }
    }
    // Far enough from every key that it was not aimed at the keyboard at all.
    var reach = bestH * 1.5;
    return bestD <= reach * reach ? best : null;
  }

  function pressFlash(el) {
    el.classList.add('down');
    setTimeout(function () { el.classList.remove('down'); }, 90);
  }
  function pressAt(x, y) {
    var el = nearestKey(x, y);
    if (el && el._wfPress) { el._wfPress(); pressFlash(el); }
  }

  // ---- rendering --------------------------------------------------------
  var GRID_N = 16;

  // The words live in a gutter around the grid rather than painted on top of
  // it. Overlaid, they hid whole tiles — and a map whose only way through a
  // wall band is in an edge column (common enough: three of the first four
  // thousand seeds) then looks completely sealed off, because the doorway is
  // underneath a word. The gutter costs a little grid size and no page height.
  var view = { cell: 1, x0: 0, y0: 0, grid: 1, pill: 12 };

  function pillSize() { return Math.max(11, canvas.width * 0.036); }

  function layout() {
    var pill = pillSize();
    var gutter = pill * 2.35;            // two stacked words, plus a little air
    var g = canvas.width - gutter * 2;
    view = { cell: g / GRID_N, x0: gutter, y0: gutter, grid: g, pill: pill, gutter: gutter };
    return view;
  }

  // The height budget is measured once and then cached, because a phone's
  // address bar collapses and expands as you scroll, which changes
  // window.innerHeight and fires `resize` — recomputing on every one of those
  // made the board jump between two sizes while scrolling. Only a genuine
  // layout change (rotation, fullscreen) re-measures it; a resize event that
  // leaves the width alone is ignored.
  var heightBudget = 0, lastStageW = -1, tallestSeen = 0;
  function resizeCanvas(remeasure) {
    var keyH = (window.innerWidth || 400) <= 400 ? 44 : 48;
    var reserve = keyH * 4 + 20 + 150;
    // Track the tallest viewport seen and budget against that. A phone's
    // address bar only ever gives height back, so the tallest reading is the
    // stable one; budgeting against whatever it happens to be right now made
    // the arena jump between sizes.
    tallestSeen = Math.max(tallestSeen, window.innerHeight || 800);
    if (remeasure || !heightBudget) heightBudget = tallestSeen - reserve;
    var stageW = stage.clientWidth;
    // Ignore the sub-pixel width wobble iOS reports while scrolling.
    if (!remeasure && Math.abs(stageW - lastStageW) <= 4) return;
    lastStageW = stageW;
    var cssW = Math.min(stageW, 560, Math.max(220, heightBudget));
    canvas.style.width = cssW + 'px';
    canvas.width = Math.round(cssW * (window.devicePixelRatio || 1));
    // Square grid plus equal gutters, and one extra row along the bottom for
    // the special — which used to sit on the middle tile and hide it.
    canvas.height = Math.round(canvas.width + pillSize() * 1.35);
    layout();
  }
  window.addEventListener('resize', function () { resizeCanvas(false); });
  window.addEventListener('orientationchange', function () { resizeCanvas(true); });
  resizeCanvas(true);

  function draw() {
    requestAnimationFrame(draw);
    if ((screen !== 'match' && screen !== 'ended') || !grid) return;
    var v = layout(), cell = v.cell, x0 = v.x0, y0 = v.y0;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    for (var y = 0; y < GRID_N; y++) {
      for (var x = 0; x < GRID_N; x++) {
        var t = grid[y][x];
        ctx.fillStyle = t === S.WALL ? '#2b2b2b' : t === S.RIVER ? '#bcdcf0' : '#fbfbf7';
        ctx.fillRect(x0 + x * cell, y0 + y * cell, cell, cell);
      }
    }
    ctx.strokeStyle = 'rgba(0,0,0,.06)';
    ctx.lineWidth = 1;
    for (var i = 0; i <= GRID_N; i++) {
      ctx.beginPath(); ctx.moveTo(x0 + i * cell, y0); ctx.lineTo(x0 + i * cell, y0 + v.grid); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(x0, y0 + i * cell); ctx.lineTo(x0 + v.grid, y0 + i * cell); ctx.stroke();
    }
    ctx.strokeStyle = 'rgba(0,0,0,.5)';
    ctx.lineWidth = 1.5;
    ctx.strokeRect(x0, y0, v.grid, v.grid);

    // my own trap: only I can see it
    if (myTrap) {
      ctx.fillStyle = 'rgba(211,84,0,.35)';
      ctx.beginPath();
      ctx.arc(x0 + (myTrap.x + .5) * cell, y0 + (myTrap.y + .5) * cell, cell * .3, 0, 7);
      ctx.fill();
    }

    var now = Date.now();
    fx = fx.filter(function (f) { return f.until > now; });

    // Attack/special sweeps flash under the characters, not over them, so
    // whoever got hit is still clearly visible on top of the flash.
    fx.forEach(function (f) {
      if (f.type !== 'tiles') return;
      ctx.fillStyle = f.color;
      f.tiles.forEach(function (t) { ctx.fillRect(x0 + t[0] * cell, y0 + t[1] * cell, cell, cell); });
    });

    // Position eases toward the real tile each frame instead of snapping,
    // so a move reads as a step. Snap once it's close enough that easing
    // further would just be sub-pixel churn.
    for (var rid in players) {
      var rp = players[rid];
      if (rp.dead || rp.hidden) continue;
      var r = renderPos[rid] || (renderPos[rid] = { x: rp.x, y: rp.y });
      var dx = rp.x - r.x, dy = rp.y - r.y;
      if (Math.abs(dx) < 0.01 && Math.abs(dy) < 0.01) { r.x = rp.x; r.y = rp.y; }
      else { r.x += dx * 0.35; r.y += dy * 0.35; }
    }

    var colorOf = {}, ci = 0;
    for (var pid in players) colorOf[pid] = COLORS[ci++ % COLORS.length];

    for (var id in players) {
      var p = players[id];
      if (p.dead || p.hidden) continue;
      var rr = renderPos[id];
      var cx = x0 + (rr.x + .5) * cell, cy = y0 + (rr.y + .5) * cell;
      var rad = cell * .3;

      // Which dot is you: a slowly breathing amber halo, drawn behind the
      // body so it never obscures the class weapon. A slightly heavier dark
      // outline alone was far too easy to lose track of mid-fight.
      if (id === myId) {
        var pulse = 0.5 + 0.5 * Math.sin(now / 340);
        ctx.strokeStyle = 'rgba(243,156,18,' + (0.10 + 0.10 * pulse).toFixed(3) + ')';
        ctx.lineWidth = Math.max(3, cell * 0.16);
        ctx.beginPath(); ctx.arc(cx, cy, rad * (1.62 + 0.10 * pulse), 0, 7); ctx.stroke();
        ctx.strokeStyle = 'rgba(243,156,18,' + (0.75 + 0.25 * pulse).toFixed(3) + ')';
        ctx.lineWidth = Math.max(2, cell * 0.075);
        ctx.beginPath(); ctx.arc(cx, cy, rad * 1.42, 0, 7); ctx.stroke();
      }

      // How much of your own vanish is left, as a ring that empties clockwise.
      // Only ever your own — an opponent's vanish is invisible by definition.
      if (id === myId && myVanishUntil > now) {
        var total = (S.CLASSES[p.cls].special.durationMs) || 3000;
        var left = Math.max(0, Math.min(1, (myVanishUntil - now) / total));
        ctx.strokeStyle = 'rgba(142,68,173,.9)';
        ctx.lineWidth = Math.max(2.5, cell * 0.1);
        ctx.beginPath();
        ctx.arc(cx, cy, rad * 1.78, -Math.PI / 2, -Math.PI / 2 + left * Math.PI * 2);
        ctx.stroke();
      }

      ctx.fillStyle = colorOf[id];
      ctx.beginPath(); ctx.arc(cx, cy, rad, 0, 7); ctx.fill();
      ctx.lineWidth = id === myId ? 2.5 : 1.5;
      ctx.strokeStyle = id === myId ? '#222' : 'rgba(0,0,0,.25)';
      ctx.stroke();
      drawFace(cx, cy, rad, DIR_ANGLE[facing[id] || 'down'], faceOf(id));
      // Stun shows how much is left, not merely that it is happening: a full
      // red ring that empties clockwise as it wears off.
      if (p.stunned || (p.stunUntil || 0) > now) {
        var span = p.stunSpan || 2000;
        var left = Math.max(0, Math.min(1, ((p.stunUntil || 0) - now) / span));
        ctx.strokeStyle = 'rgba(192,57,43,.28)';
        ctx.lineWidth = Math.max(2, cell * 0.09);
        ctx.beginPath(); ctx.arc(cx, cy, rad * 1.35, 0, 7); ctx.stroke();
        if (left > 0) {
          ctx.strokeStyle = '#c0392b';
          ctx.beginPath();
          ctx.arc(cx, cy, rad * 1.35, -Math.PI / 2, -Math.PI / 2 + left * Math.PI * 2);
          ctx.stroke();
        }
      }
      var sw = swings[id];
      var swingT = sw && now < sw.until ? (now - sw.born) / (sw.until - sw.born) : -1;
      drawAccessory(p.cls, cx, cy, rad, DIR_ANGLE[facing[id] || 'down'], swingT);
    }

    // Drawn after every body, so no dot can end up on top of a label.
    for (var tid in players) {
      var tp = players[tid];
      if (tp.dead || tp.hidden) continue;
      var tr = renderPos[tid];
      drawIdent(tp, colorOf[tid], x0 + (tr.x + .5) * cell,
                y0 + (tr.y + .5) * cell, cell, cell * .3, tid === myId);
    }

    // Damage numbers and special pulses draw last, on top of everyone.
    fx.forEach(function (f) {
      if (f.type === 'pulse') {
        var age = (now - f.born) / (f.until - f.born);
        ctx.globalAlpha = 1 - age;
        ctx.strokeStyle = f.color; ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.arc(x0 + (f.x + .5) * cell, y0 + (f.y + .5) * cell, cell * (.3 + age * .5), 0, 7);
        ctx.stroke();
        ctx.globalAlpha = 1;
      } else if (f.type === 'dmg') {
        var t = (now - f.born) / (f.until - f.born);
        ctx.globalAlpha = 1 - t;
        ctx.font = 'bold ' + Math.round(cell * .3) + 'px sans-serif';
        ctx.textAlign = 'center';
        ctx.fillStyle = f.good ? '#2e8b57' : '#c0392b';
        ctx.fillText(f.text, x0 + (f.x + .5) * cell, y0 + (f.y + .1 - t * .6) * cell);
        ctx.globalAlpha = 1;
      }
    });

    if (screen === 'ended' && endedText) drawResult(v);
    else drawLegend();
    if (reconnectNote) {
      var bh = Math.max(22, v.grid * 0.07);
      ctx.fillStyle = 'rgba(192,57,43,.92)';
      ctx.fillRect(v.x0, v.y0, v.grid, bh);
      ctx.fillStyle = '#fff';
      ctx.font = 'bold ' + Math.round(bh * 0.55) + 'px Georgia, serif';
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillText(reconnectNote, v.x0 + v.grid / 2, v.y0 + bh / 2);
    }
  }

  // Drawn over the final board rather than replacing it with a panel, so the
  // last position of everyone is still there to read.
  function drawResult(v) {
    var midY = v.y0 + v.grid / 2, h = Math.max(34, v.grid * 0.115);
    ctx.fillStyle = 'rgba(251,251,247,.94)';
    ctx.fillRect(v.x0, midY - h, v.grid, h * 2);
    ctx.strokeStyle = endedSelf ? '#2e8b57' : '#222';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.moveTo(v.x0, midY - h); ctx.lineTo(v.x0 + v.grid, midY - h);
    ctx.moveTo(v.x0, midY + h); ctx.lineTo(v.x0 + v.grid, midY + h);
    ctx.stroke();
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillStyle = endedSelf ? '#2e8b57' : '#222';
    ctx.font = 'bold ' + Math.round(h * 0.62) + 'px Georgia, serif';
    ctx.fillText(endedText, v.x0 + v.grid / 2, midY - h * 0.32);
    ctx.fillStyle = '#999';
    ctx.font = Math.round(h * 0.38) + 'px Georgia, serif';
    ctx.fillText(botMode ? 'Leave to play again' : 'Back to the lobby shortly…',
                 v.x0 + v.grid / 2, midY + h * 0.5);
  }

  // Who is who, and how hurt they are, kept inside the dot's own cell so a
  // label can never sit on top of another player: the name just above the body,
  // a health bar just below it. A bar rather than a number because you read it
  // without stopping to parse it.
  function drawIdent(p, color, cx, cy, cell, rad, mine) {
    var name = String(p.name || '?');
    if (name.length > 7) name = name.slice(0, 6) + '\u2026';
    var frac = Math.max(0, Math.min(1, p.hp / 100));

    var barW = cell * 0.74, barH = Math.max(2.5, cell * 0.1);
    var barY = cy + rad + barH * 1.15;
    // On the bottom row the bar would spill into the word gutter and sit on the
    // legend, so it flips above the body instead.
    if (barY + barH > view.y0 + view.grid - 1) barY = cy - rad - barH * 2.15;
    ctx.save();
    ctx.fillStyle = 'rgba(255,255,255,.8)';
    roundRect(cx - barW / 2, barY, barW, barH, barH / 2); ctx.fill();
    ctx.fillStyle = frac > 0.5 ? '#2e8b57' : frac > 0.25 ? '#d68910' : '#c0392b';
    if (frac > 0) { roundRect(cx - barW / 2, barY, barW * frac, barH, barH / 2); ctx.fill(); }
    ctx.strokeStyle = 'rgba(0,0,0,.22)';
    ctx.lineWidth = 0.8;
    roundRect(cx - barW / 2, barY, barW, barH, barH / 2); ctx.stroke();

    // Outlined rather than sitting on a filled pill, so it stays readable over
    // dark walls and pale floor without hiding either.
    var fs = Math.max(7, Math.round(cell * 0.27));
    ctx.font = 'bold ' + fs + 'px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'bottom';
    var ny = cy - rad - fs * 0.28;
    // Same on the top row: keep the name inside the grid rather than over the
    // words in the gutter.
    if (ny - fs > view.y0) { /* fits above */ }
    else ny = cy + rad + fs * 1.15 + (barY < cy ? 0 : barH * 1.6);
    ctx.lineWidth = Math.max(2, fs * 0.34);
    ctx.strokeStyle = 'rgba(251,251,247,.95)';
    ctx.strokeText(name, cx, ny);
    ctx.fillStyle = mine ? '#8a5000' : color;
    ctx.fillText(name, cx, ny);
    ctx.restore();
  }

  // A face per player, picked deterministically from their id so it is stable
  // for a whole match and different between players. Drawn in the same
  // facing-local frame as the weapon, so everyone looks where they are pointing.
  function faceOf(id) {
    var h = 0;
    for (var i = 0; i < id.length; i++) h = (h * 31 + id.charCodeAt(i)) % 100003;
    return { eyes: h % 3, mouth: Math.floor(h / 3) % 4 };
  }

  function drawFace(cx, cy, rad, angle, v) {
    if (rad < 6) return;               // too small to read as a face
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(angle);
    var ex = rad * 0.24, ey = rad * 0.40, er = rad * 0.20;
    for (var side = -1; side <= 1; side += 2) {
      if (v.eyes === 1) {              // squinting
        ctx.strokeStyle = 'rgba(255,255,255,.95)';
        ctx.lineWidth = Math.max(1.1, rad * 0.13);
        ctx.beginPath();
        ctx.moveTo(ex - er * 0.6, side * ey); ctx.lineTo(ex + er * 0.6, side * ey);
        ctx.stroke();
      } else {
        ctx.fillStyle = 'rgba(255,255,255,.95)';
        ctx.beginPath(); ctx.arc(ex, side * ey, er, 0, 7); ctx.fill();
        ctx.fillStyle = '#222';
        ctx.beginPath();
        ctx.arc(ex + er * 0.3, side * ey, er * (v.eyes === 2 ? 0.42 : 0.58), 0, 7);
        ctx.fill();
      }
    }
    ctx.strokeStyle = 'rgba(255,255,255,.92)';
    ctx.lineWidth = Math.max(1.1, rad * 0.12);
    var mx = rad * 0.56;
    ctx.beginPath();
    if (v.mouth === 0) { ctx.moveTo(mx, -rad * 0.15); ctx.lineTo(mx, rad * 0.15); }
    else if (v.mouth === 1) { ctx.arc(mx - rad * 0.12, 0, rad * 0.26, -Math.PI * 0.45, Math.PI * 0.45); }
    else if (v.mouth === 2) { ctx.arc(mx, 0, rad * 0.15, 0, 7); }
    else { ctx.arc(mx + rad * 0.22, 0, rad * 0.26, Math.PI * 0.55, Math.PI * 1.45); }
    ctx.stroke();
    ctx.restore();
  }

  // A plain colored dot reads as nothing in particular. Each class gets a
  // weapon silhouette drawn in local +x (facing) coordinates, then rotated to
  // the tracked facing direction — the same drawing works for all four.
  // `t` runs 0..1 through a swing, or -1 when idle. Each weapon moves the way
  // it ought to: the dagger and the wand jab straight out and back, the sword
  // sweeps through its arc, and the bow draws back before the shot leaves.
  function drawAccessory(cls, cx, cy, r, angle, t) {
    var swinging = t >= 0;
    var punch = swinging ? Math.sin(t * Math.PI) : 0;          // out and back
    var sweep = swinging ? (t - 0.5) * 1.9 : 0;                // through an arc
    var draw = swinging ? (t < 0.45 ? -t / 0.45 : -(1 - t) / 0.55) : 0;  // back, then loose
    ctx.save();
    ctx.translate(cx, cy);
    ctx.rotate(angle);
    if (cls === 'fighter') ctx.rotate(sweep);
    if (cls === 'rogue' || cls === 'mage') ctx.translate(r * 0.75 * punch, 0);
    if (cls === 'ranger') ctx.translate(r * 0.85 * draw, 0);
    if (cls === 'rogue') {
      ctx.strokeStyle = '#ddd'; ctx.lineWidth = Math.max(1.5, r * .12);
      ctx.beginPath(); ctx.moveTo(r * .9, 0); ctx.lineTo(r * 1.9, 0); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(r * .95, -r * .3); ctx.lineTo(r * .95, r * .3); ctx.stroke();
      ctx.fillStyle = '#ddd';
      ctx.beginPath(); ctx.moveTo(r * 1.9, 0); ctx.lineTo(r * 2.25, -r * .2); ctx.lineTo(r * 2.25, r * .2); ctx.closePath(); ctx.fill();
    } else if (cls === 'fighter') {
      ctx.strokeStyle = '#ccc'; ctx.lineWidth = Math.max(2, r * .16);
      ctx.beginPath(); ctx.moveTo(r * .8, 0); ctx.lineTo(r * 2.6, 0); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(r * 1.15, -r * .38); ctx.lineTo(r * 1.15, r * .38); ctx.stroke();
      ctx.fillStyle = '#ccc';
      ctx.beginPath(); ctx.arc(r * .65, 0, r * .16, 0, 7); ctx.fill();
    } else if (cls === 'ranger') {
      ctx.strokeStyle = '#a0713c'; ctx.lineWidth = Math.max(1.5, r * .12);
      ctx.beginPath(); ctx.arc(0, 0, r * 1.15, Math.PI * .6, Math.PI * 1.4); ctx.stroke();
      ctx.strokeStyle = '#654321'; ctx.lineWidth = 1;
      ctx.beginPath(); ctx.moveTo(-r * .9, 0); ctx.lineTo(r * 1.7, 0);
      ctx.moveTo(-r * .9, 0); ctx.lineTo(-r * 1.15, -r * .18);
      ctx.moveTo(-r * .9, 0); ctx.lineTo(-r * 1.15, r * .18);
      ctx.stroke();
      ctx.fillStyle = '#654321';
      ctx.beginPath(); ctx.moveTo(r * 1.7, 0); ctx.lineTo(r * 1.4, -r * .18); ctx.lineTo(r * 1.4, r * .18); ctx.closePath(); ctx.fill();
    } else if (cls === 'mage') {
      ctx.strokeStyle = '#7a5c2e'; ctx.lineWidth = Math.max(1.5, r * .12);
      ctx.beginPath(); ctx.moveTo(r * .9, 0); ctx.lineTo(r * 2.1, 0); ctx.stroke();
      ctx.fillStyle = '#9b59b6';
      ctx.beginPath(); ctx.arc(r * 2.1, 0, r * .28, 0, 7); ctx.fill();
      ctx.strokeStyle = '#d2b4de'; ctx.lineWidth = 1;
      [-1, 1].forEach(function (s) {
        ctx.beginPath(); ctx.moveTo(r * 2.1, s * r * .3); ctx.lineTo(r * 2.1, s * r * .5); ctx.stroke();
      });
    }
    ctx.restore();
  }

  // Each side of the gutter carries its pair: the move word on the outside,
  // the attack word just inside it against the grid edge — the same order on
  // all four sides, so "outer ring = walk, inner ring = swing" holds however
  // you look at it. The special goes along the bottom, under its pair.
  function drawLegend() {
    if (!myWords) return;
    var buf = typedBuf.trim().toLowerCase();
    [['moveUp', 'atkUp', 'top'], ['moveDown', 'atkDown', 'bottom'],
     ['moveLeft', 'atkLeft', 'left'], ['moveRight', 'atkRight', 'right']].forEach(function (t) {
      drawWordPill(myWords[t[0]], buf, t[2], 1, false, 'move');
      drawWordPill(myWords[t[1]], buf, t[2], 0, false, 'atk');
    });
    drawWordPill(myWords.special, buf, 'special', 0, mySpecialReadyAt > Date.now(), 'special');
  }

  // The firable word is drawn full size in its colour; the two queued behind it
  // trail smaller and greyer, close enough to read at a glance while you are
  // still typing the first. Reading ahead is the point — it is what makes a
  // three-letter word actually cheaper than a six-letter one.
  function tailPresent(list) { return !!(list && list.length); }

  function drawWordPill(queue, buf, side, offset, faded, kind) {
    if (!queue || !queue.length) return;
    var word = queue[0];
    if (!word) return;
    var matchLen = 0;
    while (matchLen < word.length && matchLen < buf.length &&
           word[matchLen].toLowerCase() === buf[matchLen]) matchLen++;
    var mismatched = buf.length > matchLen;

    var v = view, h = v.pill, g = v.grid;
    var dist = h * (0.62 + offset * 1.13);
    var x, y, rot = 0;
    if (side === 'top') { x = v.x0 + g / 2; y = v.y0 - dist; }
    else if (side === 'bottom') { x = v.x0 + g / 2; y = v.y0 + g + dist; }
    else if (side === 'special') { x = v.x0 + g / 2; y = v.y0 + g + h * 2.95; }
    else if (side === 'left') { x = v.x0 - dist; y = v.y0 + g / 2; rot = -Math.PI / 2; }
    else { x = v.x0 + g + dist; y = v.y0 + g / 2; rot = Math.PI / 2; }

    var mainPx = Math.round(h * 0.72);
    var queuePx = Math.max(7, Math.round(h * 0.5));
    var mainFont = 'bold ' + mainPx + 'px monospace';
    var queueFont = queuePx + 'px monospace';
    var upcoming = queue.slice(1).filter(Boolean);

    ctx.save();
    ctx.translate(x, y);
    if (rot) ctx.rotate(rot);
    ctx.textBaseline = 'middle';
    ctx.textAlign = 'left';

    // Centre the whole run — word plus its tail — on the gutter line.
    ctx.font = mainFont;
    var mainW = ctx.measureText(word).width;
    ctx.font = queueFont;
    // Space the tail off in pixels rather than with space characters, so the
    // gap does not depend on how the font treats runs of whitespace.
    var gap = tailPresent(upcoming) ? mainPx * 0.45 : 0;
    var tail = upcoming.join('   ');
    var tailW = tail ? ctx.measureText(tail).width + gap : 0;
    var startX = -(mainW + tailW) / 2;

    var baseColor = faded ? '#bbb' : kind === 'move' ? '#2980b9' : kind === 'atk' ? '#c0392b' : '#222';
    ctx.font = mainFont;
    if (matchLen > 0 && !mismatched) {
      var head = word.slice(0, matchLen), rest = word.slice(matchLen);
      ctx.fillStyle = faded ? '#999' : '#2e8b57';
      ctx.fillText(head, startX, 1);
      var headW = ctx.measureText(head).width;
      ctx.fillStyle = baseColor;
      ctx.fillText(rest, startX + headW, 1);
    } else {
      ctx.fillStyle = baseColor;
      ctx.fillText(word, startX, 1);
    }

    if (tail) {
      ctx.font = queueFont;
      ctx.fillStyle = faded ? 'rgba(180,180,180,.6)' : 'rgba(120,120,120,.65)';
      ctx.fillText(tail, startX + mainW + gap, 1);
    }
    ctx.restore();
  }

  function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }
