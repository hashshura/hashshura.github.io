Nothing happens until you finish typing a word, so there is no continuous
motion to desync and nothing to lag-compensate. The server is still
authoritative — it is the only thing that knows what your nine words
currently are — but the whole match is just short, discrete,
correctly-ordered events. That is the entire pitch: a battle-royale grid
brawler you can play over a genuinely bad connection.

Four classes, one arena generated fresh per room (walls block everyone,
rivers block walking but not arrows or beams), six players, last one standing.

  function api(path, opts) {
    return fetch(NET_URL + path, opts).then(function (r) {
      if (!r.ok) throw new Error('http ' + r.status);
      return r.json();
    });
  }
  function pingRoom(code) {
    var t0 = (window.performance || Date).now();
    return fetch(NET_URL + '/room/' + code + '/ping', { cache: 'no-store' })
      .then(function () { return Math.round((window.performance || Date).now() - t0); })
      .catch(function () { return 9999; });
  }
  function escapeHtml(s) {
    return String(s).replace(/[&<>"]/g, function (c) {
      return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c];
    });
  }

  // ---- screens --------------------------------------------------------
  var screen = 'menu';   // menu | lobby | match | ended
  function showScreen(s) {
    screen = s;
    menuEl.hidden = s !== 'menu';
    lobbyEl.hidden = s !== 'lobby';
    endedEl.hidden = s !== 'ended';
    exitBtn.hidden = s === 'menu';
    wrap.classList.toggle('with-menu', s === 'menu' || s === 'lobby' || s === 'ended');

    hintEl.hidden = s !== 'match';

  }

  // ---- connection -------------------------------------------------------
  var ws = null, myId = null, grid = null, players = {}, myWords = null;
  var mySpecialReadyAt = 0, myTrap = null, lastEvent = null, lastEventAt = 0;
  var typedBuf = '';

  function connect(code, password) {
    var name = (nameEl.value || 'you').slice(0, 12);
    try { localStorage.setItem('wf_name', name); } catch (e) {}
    noteEl.textContent = 'Connecting to ' + code + '…';
    var url = NET_URL.replace(/^http/, 'ws') + '/room/' + code + '/ws?name=' +
              encodeURIComponent(name) + '&cls=' + encodeURIComponent(myCls) +
              '&pw=' + encodeURIComponent(password || '');
    ws = new WebSocket(url);
    ws.onopen = function () { showScreen('lobby'); startHeartbeat(); };
    ws.onmessage = function (ev) {
      var msg;
      try { msg = JSON.parse(ev.data); } catch (e) { return; }
      if (msg.type === 'lobby') onLobby(msg);
      else if (msg.type === 'start') onStart(msg);
      else if (msg.type === 'state') onState(msg);
      else if (msg.type === 'ended') onEnded(msg);
    };
    ws.onclose = function () {
      ws = null;
      stopHeartbeat();
      if (screen !== 'menu') { showScreen('menu'); noteEl.textContent = 'Disconnected.'; }
    };
    ws.onerror = function () { noteEl.textContent = 'Could not connect.'; };
  }

  function send(obj) { if (ws && ws.readyState === 1) ws.send(JSON.stringify(obj)); }

  // Sitting in the lobby, or just thinking through a hard word, can easily
  // take longer than a real player's connection should be judged idle for —
  // this keeps the server's per-connection "last seen" fresh regardless of
  // how much actual gameplay is happening.
  var heartbeatTimer = null;
  function startHeartbeat() {
    stopHeartbeat();
    heartbeatTimer = setInterval(function () { send({ type: 'ping' }); }, 8000);
  }
  function stopHeartbeat() {
    if (heartbeatTimer) { clearInterval(heartbeatTimer); heartbeatTimer = null; }
  }

  // ---- lobby --------------------------------------------------------------
  var iAmReady = false;
  function onLobby(msg) {
    if (screen === 'match' || screen === 'ended') showScreen('lobby');
    document.getElementById('wf-lobby-title').textContent =
      msg.room ? msg.room.name + ' (' + msg.room.code + ')' : 'Room';
    rosterEl.innerHTML = '';
    msg.roster.forEach(function (p) {
      var row = document.createElement('div');
      row.className = 'p' + (p.ready ? ' ready' : '');
      row.innerHTML = '<span>' + (EMOJI[p.cls] || '') + ' ' + escapeHtml(p.name) + '</span>' +
        '<span>' + (p.ready ? '✅ ready' : '…') + '</span>';
      rosterEl.appendChild(row);
    });
    if (msg.phase === 'countdown' && msg.countdownEndsAt) {
      tickCountdown(msg.countdownEndsAt);
    } else {
      countdownEl.textContent = msg.roster.length < 2 ? 'Waiting for another player…' : '';
    }
  }
  var countdownTimer = null;
  function tickCountdown(endsAt) {
    if (countdownTimer) clearInterval(countdownTimer);
    var tick = function () {
      var left = Math.max(0, endsAt - Date.now());
      countdownEl.textContent = 'Starting in ' + Math.ceil(left / 1000) + 's…';
      if (left <= 0) clearInterval(countdownTimer);
    };
    tick();
    countdownTimer = setInterval(tick, 200);
  }
  readyBtn.addEventListener('click', function () {
    iAmReady = !iAmReady;
    readyBtn.textContent = iAmReady ? 'Cancel' : 'Ready';
    send({ type: 'ready', ready: iAmReady });
  });

  // ---- match --------------------------------------------------------------
  function onStart(msg) {
    grid = msg.grid;
    players = {};
    myWords = null; myTrap = null; lastEvent = null;
    iAmReady = false; readyBtn.textContent = 'Ready';
    if (countdownTimer) clearInterval(countdownTimer);
    typedEl.value = ''; typedBuf = '';
    showScreen('match');
    typedEl.disabled = false;
    typedEl.focus();
  }

  function onState(msg) {
    myId = msg.you;
    players = msg.players;
    if (msg.words) myWords = msg.words;
    mySpecialReadyAt = msg.specialReadyAt || 0;
    myTrap = msg.trap || null;
    if (msg.event) { lastEvent = msg.event; lastEventAt = (window.performance || Date).now(); }
    var me = players[myId];
    spectatingEl.hidden = !(me && me.dead);
    if (me && me.dead) { typedEl.disabled = true; typedEl.blur(); }
    renderScore();
  }

  function onEnded(msg) {
    var w = players[msg.winner];
    document.getElementById('wf-ended-title').textContent =
      w ? (EMOJI[w.cls] || '') + ' ' + w.name + ' wins!' : 'No one left standing';
    showScreen('ended');
    typedEl.disabled = true;
  }

  function renderScore() {
    scoreEl.innerHTML = '';
    for (var id in players) {
      var p = players[id];
      var el = document.createElement('div');
      el.className = 'who' + (id === myId ? ' self' : '') + (p.dead ? ' dead' : '');
      el.innerHTML = (EMOJI[p.cls] || '') + ' ' + escapeHtml(p.name) +
        ' <span class="hp">' + p.hp + '</span>';
      scoreEl.appendChild(el);
    }
  }

  // ---- typing ---------------------------------------------------------
  // The buffer decides its own target slot: whichever of my 9 live words it
  // exactly matches. Space/Enter submits; a non-match just clears, no penalty
  // beyond the time spent typing it.
  typedEl.addEventListener('keydown', function (ev) {
    if (ev.key === ' ' || ev.key === 'Enter') {
      ev.preventDefault();
      submitTyped();
    }
  });
  typedEl.addEventListener('input', function () { typedBuf = typedEl.value; });

  function submitTyped() {
    var word = typedBuf.trim().toLowerCase();
    typedEl.value = ''; typedBuf = '';
    if (!word || !myWords) return;
    for (var slot in myWords) {
      if (myWords[slot].toLowerCase() === word) { send({ type: 'action', slot: slot, word: word }); return; }
    }
    // no match: nothing happens, buffer is already cleared
  }

  // ---- rendering --------------------------------------------------------
  var GRID_N = 16;
  function resizeCanvas() {
    var cssW = Math.min(stage.clientWidth, 560);
    canvas.style.width = cssW + 'px';
    canvas.width = Math.round(cssW * (window.devicePixelRatio || 1));
    canvas.height = canvas.width;
  }
  window.addEventListener('resize', resizeCanvas);
  resizeCanvas();

  function draw() {
    requestAnimationFrame(draw);
    if (screen !== 'match' || !grid) return;
    var cell = canvas.width / GRID_N;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    for (var y = 0; y < GRID_N; y++) {
      for (var x = 0; x < GRID_N; x++) {
        var t = grid[y][x];
        ctx.fillStyle = t === S.WALL ? '#2b2b2b' : t === S.RIVER ? '#bcdcf0' : '#fbfbf7';
        ctx.fillRect(x * cell, y * cell, cell, cell);
      }
    }
    ctx.strokeStyle = 'rgba(0,0,0,.06)';
    for (var i = 0; i <= GRID_N; i++) {
      ctx.beginPath(); ctx.moveTo(i * cell, 0); ctx.lineTo(i * cell, canvas.height); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(0, i * cell); ctx.lineTo(canvas.width, i * cell); ctx.stroke();
    }

    // my own trap: only I can see it
    if (myTrap) {
      ctx.fillStyle = 'rgba(211,84,0,.35)';
      ctx.beginPath();
      ctx.arc((myTrap.x + .5) * cell, (myTrap.y + .5) * cell, cell * .3, 0, 7);
      ctx.fill();
    }

    var colorOf = {}, ci = 0;
    for (var pid in players) colorOf[pid] = COLORS[ci++ % COLORS.length];

    for (var id in players) {
      var p = players[id];
      if (p.dead || p.hidden) continue;
      var cx = (p.x + .5) * cell, cy = (p.y + .5) * cell;
      ctx.fillStyle = colorOf[id];
      ctx.beginPath(); ctx.arc(cx, cy, cell * .34, 0, 7); ctx.fill();
      if (id === myId) { ctx.lineWidth = 2; ctx.strokeStyle = '#222'; ctx.stroke(); }
      if (p.stunned) { ctx.strokeStyle = '#c0392b'; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(cx, cy, cell * .44, 0, 7); ctx.stroke(); }
      ctx.font = Math.round(cell * .42) + 'px sans-serif';
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillText(EMOJI[p.cls] || '?', cx, cy + 1);
    }

    drawLegend(cell);
  }

  // Overlay text lives on the map itself rather than in a separate strip, so
  // it costs no extra vertical space on a phone with the keyboard docked.
  function drawLegend(cell) {
    if (!myWords) return;
    var buf = typedBuf.trim().toLowerCase();
    var pairs = [
      { move: 'moveUp', atk: 'atkUp', side: 'top' },
      { move: 'moveDown', atk: 'atkDown', side: 'bottom' },
      { move: 'moveLeft', atk: 'atkLeft', side: 'left' },
      { move: 'moveRight', atk: 'atkRight', side: 'right' }
    ];
    pairs.forEach(function (pr) {
      drawWordPill(myWords[pr.move], buf, pr.side, 0, cell);
      drawWordPill(myWords[pr.atk], buf, pr.side, 1, cell);
    });
    var specialReady = mySpecialReadyAt <= Date.now();
    drawWordPill(myWords.special, buf, 'center', 0, cell, !specialReady);
  }

  function drawWordPill(word, buf, side, offset, cell, faded) {
    if (!word) return;
    var matchLen = 0;
    while (matchLen < word.length && matchLen < buf.length &&
           word[matchLen].toLowerCase() === buf[matchLen]) matchLen++;
    var mismatched = buf.length > matchLen;

    ctx.save();
    ctx.font = 'bold ' + Math.max(9, Math.round(cell * .32)) + 'px monospace';
    var w = ctx.measureText(word).width + 12;
    var h = Math.max(14, cell * .34);
    var x, y, rot = 0;
    if (side === 'top') { x = canvas.width / 2; y = 8 + h / 2 + offset * (h + 3); }
    else if (side === 'bottom') { x = canvas.width / 2; y = canvas.height - 8 - h / 2 - (1 - offset) * (h + 3); }
    else if (side === 'center') { x = canvas.width / 2; y = canvas.height / 2 - 2; }
    else if (side === 'left') { x = 8 + h / 2 + offset * (h + 3); y = canvas.height / 2; rot = -Math.PI / 2; }
    else { x = canvas.width - 8 - h / 2 - (1 - offset) * (h + 3); y = canvas.height / 2; rot = Math.PI / 2; }

    ctx.translate(x, y);
    if (rot) ctx.rotate(rot);
    ctx.fillStyle = faded ? 'rgba(255,255,255,.55)' : 'rgba(255,255,255,.82)';
    roundRect(-w / 2, -h / 2, w, h, 5); ctx.fill();
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';

    if (matchLen > 0 && !mismatched) {
      var head = word.slice(0, matchLen), tail = word.slice(matchLen);
      var headW = ctx.measureText(head).width, tailW = ctx.measureText(tail).width;
      var startX = -(headW + tailW) / 2;
      ctx.fillStyle = faded ? '#999' : '#c0392b';
      ctx.textAlign = 'left';
      ctx.fillText(head, startX, 1);
      ctx.fillStyle = faded ? '#bbb' : '#222';
      ctx.fillText(tail, startX + headW, 1);
      ctx.textAlign = 'center';
    } else {
      ctx.fillStyle = faded ? '#bbb' : '#222';
      ctx.fillText(word, 0, 1);
    }
    ctx.restore();
  }

  function roundRect(x, y, w, h, r) {
    ctx.beginPath();
    ctx.moveTo(x + r, y);
    ctx.arcTo(x + w, y, x + w, y + h, r);
    ctx.arcTo(x + w, y + h, x, y + h, r);
    ctx.arcTo(x, y + h, x, y, r);
    ctx.arcTo(x, y, x + w, y, r);
    ctx.closePath();
  }
  requestAnimationFrame(draw);

  // ---- menu / room list ---------------------------------------------------
  function refreshRooms() {
    roomsEl.hidden = false;
    roomsEl.innerHTML = '<div>looking for rooms…</div>';
    api('/lobby/list').then(function (data) {
      var rooms = (data.rooms || []).slice(0, 12);
      if (!rooms.length) { roomsEl.innerHTML = '<div>no rooms yet — create one?</div>'; return; }
      return Promise.all(rooms.map(function (r) {
        return pingRoom(r.code).then(function (ms) { r.ping = ms; return r; });
      })).then(function (list) {
        list.sort(function (a, b) { return a.ping - b.ping; });
        roomsEl.innerHTML = '';
        list.forEach(function (r) {
          var row = document.createElement('div');
          row.innerHTML = '<span>' + (r.private ? '🔒 ' : '') + r.code + ' · ' + escapeHtml(r.name) + '</span>' +
            '<span>' + r.players + '/' + r.max + ' · ' + (r.ping > 3000 ? '—' : r.ping + 'ms') + '</span>';
          row.addEventListener('click', function () {
            if (r.players >= r.max) { noteEl.textContent = 'That room is full.'; return; }
            var pw = r.private ? (window.prompt('Password for room ' + r.code) || '') : '';
            if (r.private && !pw) return;
            connect(r.code, pw);
          });
          roomsEl.appendChild(row);
        });
      });
    }).catch(function () { roomsEl.innerHTML = '<div>the server is not answering.</div>'; });
  }
  refreshBtn.addEventListener('click', refreshRooms);

  if (!NET_URL) {
    createBtn.disabled = true;
    noteEl.textContent = 'Online rooms are off — no server is configured.';
  } else {
    createBtn.addEventListener('click', function () {
      var rn = window.prompt('Room name:', (nameEl.value || 'our') + "'s room");
      if (rn === null) return;
      var pw = window.prompt('Password (leave blank for a public room):', '') || '';
      noteEl.textContent = 'Creating room…';
      api('/lobby/create', {
        method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: rn, password: pw })
      }).then(function (res) {
        if (res.error) { noteEl.textContent = res.error; return; }
        connect(res.code, pw);
      }).catch(function () { noteEl.textContent = 'Could not create the room.'; });
    });
  }

  exitBtn.addEventListener('click', function () {
    if (ws) { try { ws.close(); } catch (e) {} ws = null; }
    if (countdownTimer) clearInterval(countdownTimer);
    showScreen('menu');
    noteEl.textContent = '';
  });
})();
